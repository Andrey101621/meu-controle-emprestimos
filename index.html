/* ===================================================== WHATSAPP ===================================================== */ function abrirWhatsApp(indice){ const e= emprestimos[indice]; if(!e.telefone){ alert( "Telefone não cadastrado." ); return; } let telefone= e.telefone.replace( /\D/g, "" ); if( telefone.length===10|| telefone.length===11 ){ telefone="55"+telefone; } const calc= calcularEmprestimo(e); const pix= configuracoes.pix|| "Não configurado"; /* ===================================================== EMPRÉSTIMO MENSAL ===================================================== */ if(e.tipo==="mensal"){ /* Valor para renovação normalmente é somente os juros. Se houver atraso, a multa acumulada é somada automaticamente ao valor para renovação. */ const valorRenovarComMulta= calc.valorRenovar+ calc.multa; /* Mensagem mensal */ let mensagem= `Olá, ${e.cliente}! 📅 Hoje vence o seu contrato. 💵 Valor total: ${dinheiro(e.valor)} 📈 Juros: ${dinheiro(calc.juros)} Para renovar: `; /* Se estiver atrasado, informa juros + multa. Se não estiver atrasado, informa somente juros. */ if(calc.dias>0){ mensagem+= ` 📈 Somente juros + multa: ${dinheiro(valorRenovarComMulta)} ⚠️ Dias de atraso: ${calc.dias} ⚠️ Multa acumulada: ${dinheiro(calc.multa)} `; }else{ mensagem+= ` 📈 Somente juros: ${dinheiro(calc.valorRenovar)} `; } /* Multa diária sempre aparece, pois é a multa que será aplicada caso haja atraso. */ mensagem+= ` ⚠️ Multa diária após atraso: ${dinheiro(calc.multaDiaria)} 📅 Vencimento: ${formatarData(e.vencimento)} Pix para pagamento (CNPJ): ${pix}`; window.open( "https://wa.me/"+ telefone+ "?text="+ encodeURIComponent( mensagem ), "_blank" ); return; } /* ===================================================== EMPRÉSTIMO PARCELADO ===================================================== */ if(e.tipo==="parcelado"){ /* Valor da parcela + multa acumulada. Quando não existe atraso, a multa é zero, portanto o valor permanece igual ao da parcela. */ const valorParcelaComMulta= e.valorParcela+ calc.multa; /* Número da parcela atual. */ const parcelaAtual= Math.min( Number(e.parcelasRecebidas||0)+1, Number(e.totalParcelas||0) ); /* Mensagem parcelada */ let mensagem= `Olá, ${e.cliente}! 📅 Sua parcela vence hoje. 📆 Parcela: ${parcelaAtual}/${e.totalParcelas} 💵 Valor: ${dinheiro(valorParcelaComMulta)} ⚠️ Multa diária: ${dinheiro(calc.multaDiaria)} `; /* Se estiver atrasado, acrescenta a multa acumulada que já está incluída no valor enviado. */ if(calc.dias>0){ mensagem+= ` ⚠️ Dias de atraso: ${calc.dias} ⚠️ Multa acumulada: ${dinheiro(calc.multa)} 💰 Valor com multa: ${dinheiro(valorParcelaComMulta)} `; } mensagem+= ` 📅 Vencimento: ${formatarData(e.vencimento)} Pix para pagamento (CNPJ): ${pix}`; window.open( "https://wa.me/"+ telefone+ "?text="+ encodeURIComponent( mensagem ), "_blank" ); return; } }
